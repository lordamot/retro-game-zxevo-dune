table landscape
# The 15 landscape types, 28 bytes each (S3, S6, S8).
# The record starts at $06CCAA, not $06CCAE: the region is
# $06CCAA-$06CE4D, 15 x 28 exactly, ending where the movement-letter
# pointers begin; $06CCAE is the movementSpeed array at +4, which is
# why S3 and S8 count their offsets from there (S3's +$06 is +$0A
# here) and S6 counts from $06CCAA.  +0 and +2 are read by nothing
# the specs name.
rom $06CCAA
count 15
size $1C
prefix LS_
idprefix LST_
field $00 hw     f00
field $02 hw     f02
field $04 b*6    movementSpeed          ; S3: by movementType; 0 impassable
field $0A w      letUnitWobble          ; S3
field $0C w      isValidForStructure    ; S3/S6/S8: a building may stand here
field $0E w      isSand                 ; S3/S8
field $10 w      isValidForStructure2   ; S3/S6/S8: a slab may go here
field $12 sw     spice                  ; S3/S5: -1 sand, dunes and bloom, 0 rock, 25 spice, 100 thick
field $14 w      craterType             ; S8: 0 none, 1 sand, 2 rock (through tbl_crater)
field $16 sw     radarColour            ; S3/S8: the PC's; unused on the Mega Drive
field $18 w      spriteID               ; S3/S8: the PC's; unused on the Mega Drive
field $1A w      f1A
