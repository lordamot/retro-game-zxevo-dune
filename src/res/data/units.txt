# units.txt - the 27 unit types, the table at $06BC00

# Written once by tools/sega/port_data.py from orig/dune2.gen;

# The format is in the header of src/res/data/tables.txt.

table unit_info
# The 27 unit types (S2-S7).  ROM pointers to each record are at
# $06C5B4; here index a record as tbl_unit_info + UNIT_INFO_SIZE * type.
rom $06BC00
count 27
size $5C
prefix UI_
idprefix UNIT_
field $00 w      f00                    ; PC stringID_abbrev
field $02 str    name                   ; the name (units.txt, buildings.txt)
field $06 w      f06                    ; PC stringID_full
field $08 str    wsa                    ; the PC picture file it was drawn from
field $0C hw     objectFlags            ; S4/S1: bit 6 hasTurret, 11 scriptNoSlowdown, 12 targetAir, 13 priority
field $0E w      spawnChance            ; S4: chance in 256 that a Soldier survives the unit
field $10 w      hitpoints              ; S3/S4
field $12 w      sight                  ; S8: how far round it the fog lifts
field $14 w      f14                    ; PC spriteID
field $16 w      buildCredits           ; S4/S6: the cost
field $18 w      buildTime              ; S6: in build ticks
field $1A w      availableCampaign      ; S6 (ObjectInfo)
field $1C hl     structuresRequired     ; S6: a bit per structure type
field $20 b      sortPriority           ; S6/S10: the build panel's order
field $21 b      upgradeLevelRequired   ; S6
field $22 w      f22                    ; PC actionsPlayer[0]
field $24 w      f24                    ; PC actionsPlayer[1]
field $26 w      f26                    ; PC actionsPlayer[2]
field $28 w      actionPlayer           ; S7: the player's units' default order
field $2A w      f2A
field $2C w      f2C                    ; PC hintStringID
field $2E w      priorityBuild          ; S4/S6/S7
field $30 w      priorityTarget         ; S4/S6
field $32 hb     availableHouse         ; S6: a bit per house
field $33 b      f33
field $34 w      indexStart             ; S2: first unit slot of the type
field $36 w      indexEnd               ; S2: last unit slot of the type
field $38 hw     flags                  ; S3/S4: 1 isBullet, 2 explodeOnDeath, 3 sonicProtection, 4 canWobble, 5 isTracked, 6 isGroundUnit, 7 mustStayInMap, 10 firesTwice, 11 impactOnSand, 12 isNotDeviatable, 13 hasAnimationSet, 14 notAccurate, 15 isNormalUnit
field $3A w      dimension              ; S3/S8: sprite size; dimension + 3 pixels of squares
field $3C w      f3C
field $3E w      movementType           ; S3: 0 foot, 1 tracked, 2 harvester, 3 wheeled, 4 winger, 5 slither
field $40 w      animationSpeed         ; S1/S3
field $42 w      movingSpeedFactor      ; S3
field $44 w      turningSpeed           ; S3
field $46 w      f46                    ; units.txt 'sprite'; PC groundSpriteID
field $48 sw     f48                    ; units.txt 'turret' sprite, -1 none; PC turretSpriteID
field $4A sw     actionAI               ; S7: the computer's units' default order
field $4C w      f4C                    ; PC displayMode
field $4E w      f4E                    ; PC destroyedSpriteID
field $50 w      fireDelay              ; S4: reload is twice this
field $52 w      fireDistance           ; S4: in squares
field $54 w      damage                 ; S4
field $56 sw     explosionType          ; S3/S4
field $58 sw     bulletType             ; S4: the projectile unit type, -1 none
field $5A sw     bulletSound            ; S4/S10: an effect id (tbl_effect_song)
