table actions
# The fourteen orders (actions.txt).  The code reads records at
# $06CEF8 + 12n; the name of order n is the longword 4 bytes before.
# tbl_actions is that $06CEF8, so ACT_name is -4; tbl_actions_start
# is the name of order 0.
rom $06CEF4
count 14
size $0C
prefix ACT_
idprefix ORDER_
anchor $06CEF8
field $00 str    name                   ; the name is the longword 4 bytes before $06CEF8 + 12n
field $04 w      f00                    ; 1: may interrupt a unit already busy (Die, Destruct); PC switchType
field $06 w      f02                    ; 1 Attack/Move, 3 the rest; PC selectionType
field $08 w      f04                    ; PC soundID (a feedback voice id)
field $0A hw     f06                    ; PC stringID
