field $00 w      f00                    ; PC stringID_abbrev

field $06 w      f06                    ; PC stringID_full
field $08 str    wsa                    ; the PC picture file it was drawn from
field $0C hw     flags                  ; S6/S3/S5: bit 1 factory, 3 a slab (MD), 4 busyStateIsIncoming, 7 conquerable
field $0E w      spawnChance            ; S4: chance at +$E that a Soldier is left
field $10 w      hitpoints              ; S5/S6
field $12 w      sight                  ; S8
field $14 w      f14                    ; PC spriteID
field $16 w      buildCredits           ; S6
field $18 w      buildTime              ; S6: in build ticks
field $1A w      availableCampaign      ; S6: offered from campaign n - 1
field $1C hl     structuresRequired     ; S6: a longword; bit 18 the Outpost
field $20 b      sortPriority           ; S6
field $21 b      upgradeLevelRequired   ; S6
field $22 w      f22                    ; PC actionsPlayer[0]
field $24 w      f24                    ; PC actionsPlayer[1]
field $26 w      f26                    ; PC actionsPlayer[2]
field $28 w      f28                    ; PC actionsPlayer[3]
field $2A w      f2A
field $2C w      f2C                    ; PC hintStringID
field $2E w      priorityBuild          ; S6
field $30 w      priorityTarget         ; S6
field $32 hb     availableHouse         ; S6: a bit per house
field $33 b      f33
field $34 hl     enterFilter            ; S3: a bit per unit type that may enter
field $38 w      creditsStorage         ; S5
field $3A sw     powerUsage             ; S5: negative produces
field $3C w      layout                 ; S6/S8: index into the tbl_layout_* tables
field $3E w      f3E                    ; PC iconGroup
field $40 ref*3  animation:struct_anim  ; S6: three animation scripts, word index into tbl_struct_anim
field $4C sw*10  buildableUnits         ; S6: -1 unused; read for Barracks and WOR
field $60 w*3    upgradeCampaign        ; S6
